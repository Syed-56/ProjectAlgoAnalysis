PSEUDOCODE: Count Inversions using Merge Sort

FUNCTION CountInversions(A, temp, left, right) RETURNS count
------------------------------------------------
    // Base case
    IF left >= right THEN
        RETURN 0
    END IF

    mid <- left + (right - left) / 2        
    count <- 0

    // Divide + Conquer
    count <- count + CountInversions(A, temp, left, mid)
    count <- count + CountInversions(A, temp, mid + 1, right)

    // Combine
    i <- left
    j <- mid + 1
    k <- left

    WHILE i <= mid AND j <= right DO
        IF A[i] <= A[j] THEN
            temp[k] <- A[i]
            i <- i + 1
        ELSE
            temp[k] <- A[j]
            j <- j + 1
            count <- count + (mid - i + 1)  
        END IF
        k <- k + 1
    END WHILE

    WHILE i <= mid DO                     
        temp[k] <- A[i]
        i <- i + 1
        k <- k + 1
    END WHILE

    WHILE j <= right DO                    
        temp[k] <- A[j]
        j <- j + 1
        k <- k + 1
    END WHILE

    FOR x <- left TO right DO              
        A[x] <- temp[x]
    END FOR

    RETURN count
END FUNCTION


MAIN PROGRAM
------------------------------------------------
    READ n

    IF n < 1 THEN
        PRINT "Input size must be positive."
        EXIT with error
    END IF

    CREATE arrays A and temp, both of size n
    SEED random generator (mt19937_64) with 54432
    FOR i <- 0 TO n - 1 DO
        A[i] <- random integer in [0, 1,000,000,000]
    END FOR

    inversions <- CountInversions(A, temp, 0, n - 1)

    PRINT "Input size: ", n
    PRINT "Random seed: 54432"
    PRINT "Number of inversions: ", inversions
END


ANALYSIS
------------------------------------------------
Recurrence:   T(n) = 2T(n/2) + Theta(n)
Time:         Theta(n log n)
Space:        O(n) for temp array + O(log n) recursion stack
